cmake_minimum_required(VERSION 3.20)
project(WqlTools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(wql_core STATIC src/wql.cpp)
target_include_directories(wql_core PUBLIC src)
if(MSVC)
  target_compile_options(wql_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(wql_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(wql_tests tests/core_tests.cpp)
  target_link_libraries(wql_tests PRIVATE wql_core)
  add_test(NAME wql_core COMMAND wql_tests)
endif()
if(WIN32)
  add_library(WqlTools SHARED src/plugin.cpp src/WqlTools.def)
  target_include_directories(WqlTools PRIVATE vendor)
  target_link_libraries(WqlTools PRIVATE wql_core user32)
  target_compile_definitions(WqlTools PRIVATE UNICODE _UNICODE)
  set_target_properties(WqlTools PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(WqlTools PRIVATE /W4 /utf-8)
  endif()
  if(BUILD_TESTING)
    add_executable(plugin_smoke tests/plugin_smoke.cpp)
    target_include_directories(plugin_smoke PRIVATE vendor)
    target_compile_definitions(plugin_smoke PRIVATE UNICODE _UNICODE)
    add_test(NAME plugin_exports COMMAND plugin_smoke $<TARGET_FILE:WqlTools>)
  endif()
  install(TARGETS WqlTools RUNTIME DESTINATION WqlTools LIBRARY DESTINATION WqlTools)
  install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION WqlTools)
  install(DIRECTORY syntax/ DESTINATION WqlTools/syntax)
endif()
